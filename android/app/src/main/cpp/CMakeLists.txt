cmake_minimum_required(VERSION 3.22.1)
project(kazumi_gpu LANGUAGES CXX)
add_library(kazumi_gpu SHARED video_gpu.cpp)
target_compile_features(kazumi_gpu PRIVATE cxx_std_17)
target_compile_options(kazumi_gpu PRIVATE -Wall -Wextra)
target_link_libraries(kazumi_gpu PRIVATE dl)
if(ANDROID)
  target_link_options(kazumi_gpu PRIVATE -Wl,-z,max-page-size=16384)
endif()
